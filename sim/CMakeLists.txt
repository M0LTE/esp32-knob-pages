cmake_minimum_required(VERSION 3.16)
project(desk_sim C)
set(LVGL_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../managed_components/lvgl__lvgl)
file(GLOB_RECURSE LVGL_SRC ${LVGL_DIR}/src/*.c)
add_library(lvgl STATIC ${LVGL_SRC})
target_include_directories(lvgl PUBLIC ${LVGL_DIR} ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
target_compile_options(lvgl PRIVATE -O2 -w)
file(GLOB UI_SRC ../ui/*.c ../ui/pages/*.c ../ui/fonts/*.c)
add_executable(desk_sim main.c ${UI_SRC})
target_include_directories(desk_sim PRIVATE ../ui)
target_compile_options(desk_sim PRIVATE -O2 -Wall -Wno-unused-function)
target_link_libraries(desk_sim lvgl m)
